1. Install tools
winget install Microsoft.DotNet.SDK.8 --accept-package-agreements --accept-source-agreements
dotnet tool install --global dotnet-ef --version 8.0.11

2. Create / restore project
cd d:\CHAMRERN\Banking-Management-System\BankingManagementSystem
dotnet restore
dotnet build
dotnet add package Microsoft.EntityFrameworkCore.Sqlite --version 8.0.11

3. Database (EF Core)
dotnet ef migrations add InitialCreate
dotnet ef database update

4. Run the app
dotnet run --urls http://localhost:5180

5. Check SQL Server tables (optional)
sqlcmd -S . -E -C -d BankingManagementSystem -Q "SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_TYPE='BASE TABLE' ORDER BY TABLE_NAME;"
